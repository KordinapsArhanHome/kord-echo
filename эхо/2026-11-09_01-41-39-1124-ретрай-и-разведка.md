import asyncio, json, sqlite3, shutil, os, re
from telethon import TelegramClient
cfg = json.load(open('/root/рассылка/конфиг.json', encoding='utf-8'))
API_ID, API_HASH = cfg['api_id'], cfg['api_hash']
BASE = '/root/kord-linza'
WORK = os.path.join(BASE, 'стек', '.work-read1102')
os.makedirs(WORK, exist_ok=True)
shutil.copyfile('/root/рассылка/sessions/acc1.session', os.path.join(WORK, 'acc1.session'))
db = sqlite3.connect(os.path.join(WORK, 'acc1.session'))
db.execute('update sessions set server_address=?, port=?', ('127.0.0.1', 17443))
db.commit(); db.close()
MASK = re.compile('[0-9+][0-9 -]{9,}')
OUT = '/root/kord-glubina/kordinaps-machine/проекты/ЧТЕНИЕ-credity_zalogy-2026-11-07.md'
CH = 'credity_zalogy'
async def main():
    client = TelegramClient(os.path.join(WORK, 'acc1'), API_ID, API_HASH, receive_updates=False, request_retries=1, connection_retries=1, timeout=25)
    await asyncio.wait_for(client.connect(), 35)
    if not await client.is_user_authorized():
        print('acc1 НЕ авторизован')
        await client.disconnect()
        return
    ent = await client.get_entity(CH)
    msgs = await client.get_messages(ent, limit=200)
    rows = []
    for m in reversed(msgs):
        if not m or not m.message:
            continue
        user = None; name = None
        try:
            s = await m.get_sender()
            if s:
                user = getattr(s, 'username', None)
                name = ((getattr(s, 'first_name', '') or '') + ' ' + (getattr(s, 'last_name', '') or '')).strip()
        except Exception:
            pass
        rows.append({'id': m.id, 'date': m.date.isoformat() if m.date else None, 'user': user, 'name': name, 'text': m.message[:3000]})
    await client.disconnect()
    with open(OUT, 'w', encoding='utf-8') as f:
        f.write('# ЧТЕНИЕ credity_zalogy - свежие посты (до 70), снято 07.11 утро, acc1 рабочая копия сессии\n')
        for r in rows:
            f.write('\n## пост ' + str(r['id']) + ' | ' + str(r['date']) + ' | @' + str(r['user']) + ' | ' + str(r['name']) + '\n')
            f.write((r['text'] or '') + '\n')
    print('СНЯТО ПОСТОВ:', len(rows))
    for r in rows:
        t = MASK.sub('[ТЕЛЕФОН]', (r['text'] or '').replace('\n', ' '))[:110]
        print(str(r['id']) + ' | ' + str((r['date'] or '')[:19]) + ' | @' + str(r['user']) + ' | ' + t)
asyncio.run(main())
shutil.rmtree(WORK, ignore_errors=True)
print('конец')
